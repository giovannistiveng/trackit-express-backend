import { ShipmentRepositoryPort } from '../../../domain/repositories/shipment.repository.port';
import { CreateShipmentDTO, ShipmentResponse } from '../../dtos/shipment.dto';
import { NotFoundError, ValidationError } from '../../../shared/errors';
import { OpenStreetAdapter } from '../../../infrastructure/external/open-street.adapter';
import { ZonaTarifaRepositoryPort } from '../../../domain/repositories/zona-tarifa.repository.port';

export class CreateShipmentUseCase {
  constructor(
    private shipmentRepository: ShipmentRepositoryPort,
    private zonaRepository: ZonaTarifaRepositoryPort,
    private openStreet: OpenStreetAdapter,
  ) {}

  async execute(dto: CreateShipmentDTO): Promise<ShipmentResponse> {
    // Validar que la zona exista
    const zona = await this.zonaRepository.findById(dto.id_zona);
    if (!zona) {
      throw new NotFoundError(`No existe la zona de tarifa con id ${dto.id_zona}`);
    }

    // Calcular tarifa usando OpenStreetMap
    const { distancia_km } = await this.openStreet.calcularDistancia(dto.origen, dto.destino);

    // Tarifa base de la zona + costo por kilÃ³metro
    const costoPorKm = 1000;
    const tarifa = (zona as { tarifa_base: number }).tarifa_base + (distancia_km * costoPorKm);

    // Generar PIN de seguridad de 4 dÃ­gitos
    const pin_seguridad = Math.floor(1000 + Math.random() * 9000).toString();

    const envio = await this.shipmentRepository.create({
      origen: dto.origen,
      destino: dto.destino,
      tipo_carga: dto.tipo_carga,
      tarifa: Math.round((zona.tarifa_base + distancia_km * costoPorKm) * 100) / 100,
      pin_seguridad,
      id_cliente: dto.id_cliente,
      id_zona: dto.id_zona,
    });

    return {
      id_envio: envio.id_envio!,
      origen: envio.origen,
      destino: envio.destino,
      tipo_carga: envio.tipo_carga,
      tarifa: envio.tarifa,
      pin_seguridad: envio.pin_seguridad!,
      estado: 'Pendiente',
      id_cliente: envio.id_cliente,
      id_conductor: envio.id_conductor,
      id_zona: envio.id_zona,
    };
  }
}
